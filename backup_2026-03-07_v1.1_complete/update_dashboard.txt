# 3. PUSH TO FIREBASE
# ==========================================
def update_dashboard():
    try:
        data_to_push = fetch_and_format_weekly_data()
        
        # Target the 'newsletters' collection, and the specific document named 'latest'
        print("Pushing data to Firestore...")
        doc_ref = db.collection('newsletters').document('latest')
        doc_ref.set(data_to_push)
        
        print("✅ Dashboard successfully updated in Firebase!")
        print("Refresh your live HTML page, and you will see the new numbers!")
        
    except Exception as e:
        print(f"❌ Error updating dashboard: {e}")

if __name__ == "__main__":
